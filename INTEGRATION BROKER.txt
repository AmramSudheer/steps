class BOOK_HNDLR extends PS_PT:Integration:IRequestHandler
   method OnRequest(&requestMsg As Message) Returns Message;
end-class;

method OnRequest
   /+ &requestMsg as Message +/
   /+ Returns Message +/
   /+ Extends/implements PS_PT:Integration:IRequestHandler.OnRequest +/
   
   Local Message &responseMsg;
   Local XmlDoc &xmlDoc;
   Local XmlDoc &respDoc;
   Local XmlNode &node;
   
   Local string &bookId;
   Local string &bookName;
   Local string &bookAuthor;
   Local integer &rowCount;
   
   /* Get incoming XML */
   &xmlDoc = &requestMsg.GetXmlDoc();
   
   If None(&xmlDoc) Then
      Error ("Request XML is empty.");
   End-If;
   
   /* Read BOOK_ID - FIXED: Added [1] to handle Array return type */
   &node = &xmlDoc.DocumentElement.GetElementsByTagName("BOOK_ID")[1];
   
   If None(&node) Then
      Error ("BOOK_ID is missing.");
   End-If;
   
   &bookId = &node.NodeValue;
   
   /* Read BOOK_NAME - FIXED: Added [1] to handle Array return type */
   &node = &xmlDoc.DocumentElement.GetElementsByTagName("BOOK_NAME")[1];
   
   If None(&node) Then
      Error ("BOOK_NAME is missing.");
   End-If;
   
   &bookName = &node.NodeValue;
   
   /* Read BOOK_AUTHOR - FIXED: Added [1] to handle Array return type */
   &node = &xmlDoc.DocumentElement.GetElementsByTagName("BOOK_AUTHOR")[1];
   
   If None(&node) Then
      Error ("BOOK_AUTHOR is missing.");
   End-If;
   
   &bookAuthor = &node.NodeValue;
   
   /* Check existing BOOK_ID */
   SQLExec("SELECT COUNT(*) FROM PS_BOOK_REC WHERE BOOK_ID = :1", &bookId, &rowCount);
   
   /* Update or Insert */
   If &rowCount > 0 Then
      
      SQLExec("UPDATE PS_BOOK_REC SET BOOK_NAME = :1, BOOK_AUTHOR = :2 WHERE BOOK_ID = :3", &bookName, &bookAuthor, &bookId);
      
   Else
      
      SQLExec("INSERT INTO PS_BOOK_REC (BOOK_ID, BOOK_NAME, BOOK_AUTHOR) VALUES (:1, :2, :3)", &bookId, &bookName, &bookAuthor);
      
   End-If;
   
   /* Create response XML */
   &responseMsg = CreateMessage(Operation.BOOK_OP, %IntBroker_Response);
   &respDoc = CreateXmlDoc("<?xml version=""1.0"" encoding=""UTF-8""?><BOOK_RESPONSE_STATUS>SUCCESS</BOOK_RESPONSE_STATUS>");
   &responseMsg.SetXmlDoc(&respDoc);
   
   Return &responseMsg;
   
end-method;)

 (import PS_PT:Integration:*;

class BOOK_ROW_HNDLR extends PS_PT:Integration:IRequestHandler
   method OnRequest(&requestMsg As Message) Returns Message;
end-class;


method OnRequest
   /+ &requestMsg as Message +/
   /+ Returns Message +/
   /+ Extends/implements PS_PT:Integration:IRequestHandler.OnRequest +/
   
   Local Message &responseMsg;
   Local Rowset &inRowset, &respRowset;
   Local Record &rec;
   Local integer &i;
   Local string &BOOKID;
   Local number &countRows;
   Local boolean &saved;
   
   /* Read request */
   &inRowset = &requestMsg.GetRowset();
   
   If None(&inRowset) Then
      throw CreateException(0, 0, "Request rowset is missing.");
   End-If;
   
   /* Process worker rows */
   For &i = 1 To &inRowset.ActiveRowCount
      
      &rec = &inRowset.GetRow(&i).GetRecord(Record.BOOK_REC);
      
      &BOOKID = &rec.BOOK_ID.Value;
      
      If None(&BOOKID) Then
         throw CreateException(0, 0, "BOOK_ID is missing.");
      End-If;
      
      /* Check whether worker exists */
      SQLExec("SELECT COUNT(*) FROM PS_BOOK_REC WHERE BOOK_ID = :1", &BOOKID, &countRows);
      
      If &countRows > 0 Then
         
         /* Update existing row */
         &saved = &rec.Update();
         
      Else
         
         /* Insert new row */
         &saved = &rec.Insert();
         
      End-If;
      
      If Not &saved Then
         throw CreateException(0, 0, "BOOK_REC save failed.");
      End-If;
      
   End-For;
   
   /* Return response */
   &responseMsg = CreateMessage(Operation.BOOK_OP_R, %IntBroker_Response);
   
   &respRowset = &responseMsg.GetRowset();
   &inRowset.CopyTo(&respRowset);
   
   Return &responseMsg;
   
end-method;)
